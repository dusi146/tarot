// Gọi duy nhất API OpenCode async function callAI(systemPrompt, isChat = false, chatMemory = []) { const messages = isChat ? chatMemory : [{ role: "user", content: systemPrompt }]; const res = await fetchWithRetry("https://api.opencode.ai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${OPENCODE_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: "deepseek-v4-flash-free", messages: messages, max_tokens: 1500 }) }, 3, 1000); if (!res.ok) throw new Error("OpenCode Sập"); const data = await res.json(); return data.choices[0].message.content; }
